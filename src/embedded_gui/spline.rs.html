1//! Catmull-Rom and Cardinal cubic spline curves adapted from `bevy_curve::cubic_splines`.
2//!
3//! Provides smooth interpolation passing directly through control points, ideal for
4//! sensor plots, smoothed vector paths, and continuous animation tracks in `#![no_std]`.
5
6use embedded_graphics_core::geometry::Point;
7
8/// A cubic spline evaluator using the Catmull-Rom formulation.
9///
10/// In a Catmull-Rom spline, the curve passes directly through every control point
11/// (unlike Bézier curves where intermediate control points only act as attractors).
12#[derive(Debug, Clone, Copy)]
13pub struct CatmullRomSpline<'a> {
14    points: &'a [Point],
15}
16
17impl<'a> CatmullRomSpline<'a> {
18    /// Creates a new Catmull-Rom spline from a slice of control points.
19    /// Returns `None` if the slice contains fewer than 2 points.
20    pub const fn new(points: &'a [Point]) -> Option<Self> {
21        if points.len() < 2 {
22            None
23        } else {
24            Some(Self { points })
25        }
26    }
27
28    /// Evaluates the spline at normalized parameter `t` in `[0.0, 1.0]`.
29    pub fn evaluate(&self, t: f32) -> Point {
30        self.evaluate_cardinal(t, 0.0)
31    }
32
33    /// Evaluates a Cardinal spline with a given `tension` parameter (-1.0 to 1.0, 0.0 is standard Catmull-Rom).
34    pub fn evaluate_cardinal(&self, t: f32, tension: f32) -> Point {
35        let n = self.points.len();
36        if n == 0 {
37            return Point::zero();
38        }
39        if n == 1 {
40            return self.points[0];
41        }
42
43        let t = t.clamp(0.0, 1.0);
44        let num_segments = (n - 1) as f32;
45        let scaled_t = t * num_segments;
46        let segment_idx = (scaled_t as usize).min(n - 2);
47        let seg_t = scaled_t - segment_idx as f32;
48
49        let p0 = if segment_idx == 0 {
50            // Extrapolate phantom point before start: 2 * p1 - p2
51            Point::new(
52                2 * self.points[0].x - self.points[1].x,
53                2 * self.points[0].y - self.points[1].y,
54            )
55        } else {
56            self.points[segment_idx - 1]
57        };
58
59        let p1 = self.points[segment_idx];
60        let p2 = self.points[segment_idx + 1];
61
62        let p3 = if segment_idx + 2 >= n {
63            // Extrapolate phantom point after end: 2 * p2 - p1
64            Point::new(
65                2 * self.points[n - 1].x - self.points[n - 2].x,
66                2 * self.points[n - 1].y - self.points[n - 2].y,
67            )
68        } else {
69            self.points[segment_idx + 2]
70        };
71
72        let s = (1.0 - tension) * 0.5;
73
74        let t2 = seg_t * seg_t;
75        let t3 = t2 * seg_t;
76
77        // Basis functions for Cardinal/Catmull-Rom
78        let h1 = 2.0 * t3 - 3.0 * t2 + 1.0;
79        let h2 = -2.0 * t3 + 3.0 * t2;
80        let h3 = t3 - 2.0 * t2 + seg_t;
81        let h4 = t3 - t2;
82
83        let m1_x = s * (p2.x - p0.x) as f32;
84        let m1_y = s * (p2.y - p0.y) as f32;
85        let m2_x = s * (p3.x - p1.x) as f32;
86        let m2_y = s * (p3.y - p1.y) as f32;
87
88        let x = h1 * p1.x as f32 + h2 * p2.x as f32 + h3 * m1_x + h4 * m2_x;
89        let y = h1 * p1.y as f32 + h2 * p2.y as f32 + h3 * m1_y + h4 * m2_y;
90
91        Point::new((x + 0.5) as i32, (y + 0.5) as i32)
92    }
93
94    /// Evaluates `samples` evenly distributed points along the spline into a fixed buffer.
95    pub fn sample_into(&self, buffer: &mut [Point]) {
96        if buffer.is_empty() {
97            return;
98        }
99        if buffer.len() == 1 {
100            buffer[0] = self.evaluate(0.0);
101            return;
102        }
103        let denom = (buffer.len() - 1) as f32;
104        for (i, p) in buffer.iter_mut().enumerate() {
105            *p = self.evaluate(i as f32 / denom);
106        }
107    }
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    #[test]
115    fn test_catmull_rom_passes_through_control_points() {
116        let points = [
117            Point::new(0, 0),
118            Point::new(10, 20),
119            Point::new(20, 10),
120            Point::new(30, 30),
121        ];
122        let spline = CatmullRomSpline::new(&points).unwrap();
123
124        // Start point t=0.0
125        assert_eq!(spline.evaluate(0.0), Point::new(0, 0));
126
127        // Point 1 at t=1/3
128        let p1 = spline.evaluate(1.0 / 3.0);
129        assert!((p1.x - 10).abs() <= 1 && (p1.y - 20).abs() <= 1);
130
131        // Point 2 at t=2/3
132        let p2 = spline.evaluate(2.0 / 3.0);
133        assert!((p2.x - 20).abs() <= 1 && (p2.y - 10).abs() <= 1);
134
135        // End point t=1.0
136        assert_eq!(spline.evaluate(1.0), Point::new(30, 30));
137    }
138
139    #[test]
140    fn test_catmull_rom_sampling() {
141        let points = [Point::new(0, 0), Point::new(100, 100)];
142        let spline = CatmullRomSpline::new(&points).unwrap();
143
144        let mut samples = [Point::zero(); 5];
145        spline.sample_into(&mut samples);
146
147        assert_eq!(samples[0], Point::new(0, 0));
148        assert_eq!(samples[2], Point::new(50, 50));
149        assert_eq!(samples[4], Point::new(100, 100));
150    }
151}
