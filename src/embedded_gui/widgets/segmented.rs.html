10// ---------------------------------------------------------------------------
11// 7-Segment Bitmask Constants
12// ---------------------------------------------------------------------------
13//   AAA
14//  F   B
15//  F   B
16//   GGG
17//  E   C
18//  E   C
19//   DDD   (DP)
20pub const SEG_7_A: u8 = 1 << 0;
21pub const SEG_7_B: u8 = 1 << 1;
22pub const SEG_7_C: u8 = 1 << 2;
23pub const SEG_7_D: u8 = 1 << 3;
24pub const SEG_7_E: u8 = 1 << 4;
25pub const SEG_7_F: u8 = 1 << 5;
26pub const SEG_7_G: u8 = 1 << 6;
27pub const SEG_7_DP: u8 = 1 << 7;

29/// Encodes an ASCII character into a 7-segment bitmask.
30pub const fn encode_7seg(c: char) -> u8 {
31    match c {
32        '0' => SEG_7_A | SEG_7_B | SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_F,
33        '1' => SEG_7_B | SEG_7_C,
34        '2' => SEG_7_A | SEG_7_B | SEG_7_D | SEG_7_E | SEG_7_G,
35        '3' => SEG_7_A | SEG_7_B | SEG_7_C | SEG_7_D | SEG_7_G,
36        '4' => SEG_7_B | SEG_7_C | SEG_7_F | SEG_7_G,
37        '5' => SEG_7_A | SEG_7_C | SEG_7_D | SEG_7_F | SEG_7_G,
38        '6' => SEG_7_A | SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_F | SEG_7_G,
39        '7' => SEG_7_A | SEG_7_B | SEG_7_C,
40        '8' => SEG_7_A | SEG_7_B | SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_F | SEG_7_G,
41        '9' => SEG_7_A | SEG_7_B | SEG_7_C | SEG_7_D | SEG_7_F | SEG_7_G,
42        'a' | 'A' => SEG_7_A | SEG_7_B | SEG_7_C | SEG_7_E | SEG_7_F | SEG_7_G,
43        'b' | 'B' => SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_F | SEG_7_G,
44        'c' => SEG_7_D | SEG_7_E | SEG_7_G,
45        'C' => SEG_7_A | SEG_7_D | SEG_7_E | SEG_7_F,
46        'd' | 'D' => SEG_7_B | SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_G,
47        'e' | 'E' => SEG_7_A | SEG_7_D | SEG_7_E | SEG_7_F | SEG_7_G,
48        'f' | 'F' => SEG_7_A | SEG_7_E | SEG_7_F | SEG_7_G,
49        'h' | 'H' => SEG_7_B | SEG_7_C | SEG_7_E | SEG_7_F | SEG_7_G,
50        'l' | 'L' => SEG_7_D | SEG_7_E | SEG_7_F,
51        'o' | 'O' => SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_G,
52        'p' | 'P' => SEG_7_A | SEG_7_B | SEG_7_E | SEG_7_F | SEG_7_G,
53        'r' | 'R' => SEG_7_E | SEG_7_G,
54        'u' | 'U' => SEG_7_B | SEG_7_C | SEG_7_D | SEG_7_E | SEG_7_F,
55        '-' => SEG_7_G,
56        '_' => SEG_7_D,
57        '.' => SEG_7_DP,
58        '=' => SEG_7_D | SEG_7_G,
59        '°' => SEG_7_A | SEG_7_B | SEG_7_F | SEG_7_G,
60        _ => 0,
61    }
62}

278/// Visual styling configuration for segment displays.
279#[derive(Debug, Clone, Copy, PartialEq, Eq)]
280pub struct SegmentStyle {
281    /// Color of active (illuminated) segments.
282    pub on_color: Rgb565,
283    /// Color of inactive (unlit) segments (providing an LCD/LED ghosting effect).
284    pub off_color: Option<Rgb565>,
285    /// Segment thickness in pixels.
286    pub segment_thickness: u8,
287    /// Spacing between digits in pixels.
288    pub digit_spacing: u8,
289}
290
291impl SegmentStyle {
292    pub const fn new(on_color: Rgb565) -> Self {
293        Self {
294            on_color,
295            off_color: None,
296            segment_thickness: 2,
297            digit_spacing: 4,
298        }
299    }
300
301    pub const fn with_ghosting(mut self, off_color: Rgb565) -> Self {
302        self.off_color = Some(off_color);
303        self
304    }
305}

307/// A 7-segment multi-character display widget.
308#[derive(Debug, Clone, Copy, PartialEq, Eq)]
309pub struct SevenSegmentDisplay<'a> {
310    pub text: &'a str,
311    pub style: SegmentStyle,
312}
313
314impl<'a> SevenSegmentDisplay<'a> {
315    pub const fn new(text: &'a str, style: SegmentStyle) -> Self {
316        Self { text, style }
317    }
318
319    /// Calculates bounding rectangles for each 7-segment digit bar inside `digit_rect`.
320    pub fn digit_segments(digit_rect: Rect, thickness: u8) -> [Rect; 7] {
321        let t = thickness as u32;
322        let x = digit_rect.x;
323        let y = digit_rect.y;
324        let w = digit_rect.w;
325        let h = digit_rect.h;
326        let mid_y = y + (h as i32 - t as i32) / 2;
327        let bot_y = y + h as i32 - t as i32;
328        let right_x = x + w as i32 - t as i32;
329        let half_h = (h.saturating_sub(t * 3)) / 2;
330
331        [
332            // A: Top horizontal
333            Rect::new(x + t as i32, y, w.saturating_sub(t * 2), t),
334            // B: Top-right vertical
335            Rect::new(right_x, y + t as i32, t, half_h),
336            // C: Bottom-right vertical
337            Rect::new(right_x, mid_y + t as i32, t, half_h),
338            // D: Bottom horizontal
339            Rect::new(x + t as i32, bot_y, w.saturating_sub(t * 2), t),
340            // E: Bottom-left vertical
341            Rect::new(x, mid_y + t as i32, t, half_h),
342            // F: Top-left vertical
343            Rect::new(x, y + t as i32, t, half_h),
344            // G: Middle horizontal
345            Rect::new(x + t as i32, mid_y, w.saturating_sub(t * 2), t),
346        ]
347    }
348}

350#[cfg(test)]
351mod tests {
352    use super::*;
353
354    #[test]
355    fn test_7seg_encoding() {
356        assert_eq!(encode_7seg('0'), 0x3F);
357        assert_eq!(encode_7seg('1'), 0x06);
358        assert_eq!(encode_7seg('8'), 0x7F);
359        assert_eq!(encode_7seg('-'), 0x40);
360        assert_eq!(encode_7seg(' '), 0x00);
361    }
362
363    #[test]
364    fn test_14seg_encoding() {
365        assert_ne!(encode_14seg('A'), 0);
366        assert_ne!(encode_14seg('Z'), 0);
367        assert_ne!(encode_14seg('+'), 0);
368        assert_eq!(encode_14seg(' '), 0);
369    }
370
371    #[test]
372    fn test_digit_segment_geometry() {
373        let rect = Rect::new(10, 10, 20, 40);
374        let segs = SevenSegmentDisplay::digit_segments(rect, 2);
375        assert_eq!(segs.len(), 7);
376        // Top segment A width = 20 - 4 = 16
377        assert_eq!(segs[0].w, 16);
378        assert_eq!(segs[0].h, 2);
379    }
380
381    #[test]
382    fn test_7seg_and_14seg_all_ascii_and_symbols() {
383        for c in (0u32..=127).filter_map(char::from_u32) {
384            let _ = encode_7seg(c);
385            let _ = encode_14seg(c);
386        }
387        let _ = encode_7seg('°');
388        let _ = encode_14seg('*');
389
390        for digit in '0'..='9' {
391            assert_ne!(encode_7seg(digit), 0);
392            assert_ne!(encode_14seg(digit), 0);
393        }
394    }
