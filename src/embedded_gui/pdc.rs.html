14/// Type of Precise Draw Command.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum PdcCommandType {
17    /// Arbitrary line/polygon path with standard integer coordinates.
18    Path,
19    /// Circle with center point and radius.
20    Circle,
21    /// Arbitrary path with 13.3 fixed-point subpixel precision (1/8th pixel).
22    PrecisePath,
23}
24
25/// Error indicating capacity limit exceeded during PDC construction.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub struct PdcError;

29/// A 13.3 fixed-point coordinate point (1 unit = 1/8th pixel).
30#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
31pub struct PdcPrecisePoint {
32    pub x_fixed: i16,
33    pub y_fixed: i16,
34}
35
36impl PdcPrecisePoint {
37    pub const fn from_subpixels(x_fixed: i16, y_fixed: i16) -> Self {
38        Self { x_fixed, y_fixed }
39    }
40
41    pub const fn from_pixels(x: i16, y: i16) -> Self {
42        Self {
43            x_fixed: x << 3,
44            y_fixed: y << 3,
45        }
46    }
47
48    pub const fn to_pixel_point(self) -> Point {
49        Point::new((self.x_fixed >> 3) as i32, (self.y_fixed >> 3) as i32)
50    }
51
52    pub fn to_f32_point(self) -> (f32, f32) {
53        (self.x_fixed as f32 / 8.0, self.y_fixed as f32 / 8.0)
54    }
55}

57/// A single Precise Draw Command encoding stroke, fill, and geometry.
58#[derive(Clone, Debug, PartialEq)]
59pub struct PdcCommand<const MAX_POINTS: usize = 16> {
60    pub command_type: PdcCommandType,
61    pub stroke_color: Option<Rgb565>,
62    pub fill_color: Option<Rgb565>,
63    pub stroke_width: u8,
64    pub radius: u16,
65    pub is_closed: bool,
66    pub points: Vec<PdcPrecisePoint, MAX_POINTS>,
67}
68
69impl<const MAX_POINTS: usize> PdcCommand<MAX_POINTS> {
70    pub const fn new(command_type: PdcCommandType) -> Self {
71        Self {
72            command_type,
73            stroke_color: None,
74            fill_color: None,
75            stroke_width: 1,
76            radius: 0,
77            is_closed: false,
78            points: Vec::new(),
79        }
80    }
81
82    pub fn circle(
83        center: Point,
84        radius: u16,
85        stroke: Option<Rgb565>,
86        fill: Option<Rgb565>,
87        stroke_width: u8,
88    ) -> Self {
89        let mut cmd = Self::new(PdcCommandType::Circle);
90        cmd.stroke_color = stroke;
91        cmd.fill_color = fill;
92        cmd.stroke_width = stroke_width;
93        cmd.radius = radius;
94        let p = PdcPrecisePoint::from_pixels(center.x as i16, center.y as i16);
95        let _ = cmd.points.push(p);
96        cmd
97    }
98
99    pub fn add_point(&mut self, pt: Point) -> Result<(), PdcError> {
100        self.points
101            .push(PdcPrecisePoint::from_pixels(pt.x as i16, pt.y as i16))
102            .map_err(|_| PdcError)
103    }
104
105    pub fn add_subpixel_point(&mut self, x_fixed: i16, y_fixed: i16) -> Result<(), PdcError> {
106        self.points
107            .push(PdcPrecisePoint::from_subpixels(x_fixed, y_fixed))
108            .map_err(|_| PdcError)
109    }
110
111    /// Adds a quadratic Bézier curve segment flattened into subpixel points.
112    pub fn add_bezier_quad(
113        &mut self,
114        p0: Point,
115        p1: Point,
116        p2: Point,
117        steps: usize,
118    ) -> Result<(), PdcError> {
119        let n = steps.max(2);
120        for i in 1..=n {
121            let t = i as f32 / n as f32;
122            let one_minus_t = 1.0 - t;
123            let x = one_minus_t * one_minus_t * p0.x as f32
124                + 2.0 * one_minus_t * t * p1.x as f32
125                + t * t * p2.x as f32;
126            let y = one_minus_t * one_minus_t * p0.y as f32
127                + 2.0 * one_minus_t * t * p1.y as f32
128                + t * t * p2.y as f32;
129            let x_scaled = x * 8.0;
130            let y_scaled = y * 8.0;
131            let x_sub = (x_scaled + if x_scaled >= 0.0 { 0.5 } else { -0.5 }) as i16;
132            let y_sub = (y_scaled + if y_scaled >= 0.0 { 0.5 } else { -0.5 }) as i16;
133            self.add_subpixel_point(x_sub, y_sub)?;
134        }
135        Ok(())
136    }
137
138    /// Adds a cubic Bézier curve segment flattened into subpixel points.
139    pub fn add_bezier_cubic(
140        &mut self,
141        p0: Point,
142        p1: Point,
143        p2: Point,
144        p3: Point,
145        steps: usize,
146    ) -> Result<(), PdcError> {
147        let n = steps.max(3);
148        for i in 1..=n {
149            let t = i as f32 / n as f32;
150            let one_minus_t = 1.0 - t;
151            let t_sq = t * t;
152            let omt_sq = one_minus_t * one_minus_t;
153            let x = omt_sq * one_minus_t * p0.x as f32
154                + 3.0 * omt_sq * t * p1.x as f32
155                + 3.0 * one_minus_t * t_sq * p2.x as f32
156                + t_sq * t * p3.x as f32;
157            let y = omt_sq * one_minus_t * p0.y as f32
158                + 3.0 * omt_sq * t * p1.y as f32
159                + 3.0 * one_minus_t * t_sq * p2.y as f32
160                + t_sq * t * p3.y as f32;
161            let x_scaled = x * 8.0;
162            let y_scaled = y * 8.0;
163            let x_sub = (x_scaled + if x_scaled >= 0.0 { 0.5 } else { -0.5 }) as i16;
164            let y_sub = (y_scaled + if y_scaled >= 0.0 { 0.5 } else { -0.5 }) as i16;
165            self.add_subpixel_point(x_sub, y_sub)?;
166        }
167        Ok(())
168    }
169
170    /// Renders this vector command into the provided [`RenderCtx`].
171    pub fn render<D, C>(&self, ctx: &mut RenderCtx<'_, D, C>, offset: Point) -> Result<(), D::Error>
172    where
173        D: DrawTarget<Color = Rgb565>,
174        C: crate::render::Compositor<D>,
175    {
176        match self.command_type {
177            PdcCommandType::Circle => {
178                if let Some(center_pt) = self.points.first() {
179                    let p = center_pt.to_pixel_point();
180                    let cx = p.x + offset.x;
181                    let cy = p.y + offset.y;
182                    let r = self.radius as u32;
183
184                    if let Some(fill) = self.fill_color {
185                        ctx.fill_circle(cx, cy, r, fill)?;
186                    }
187                    if let Some(stroke) = self.stroke_color {
188                        if self.stroke_width > 0 {
189                            ctx.stroke_circle(cx, cy, r, stroke)?;
190                        }
191                    }
192                }
193            }
194            PdcCommandType::Path | PdcCommandType::PrecisePath => {
195                if self.points.len() < 2 {
196                    return Ok(());
197                }
198
199                // Render stroke lines connecting points
200                if let Some(stroke) = self.stroke_color {
201                    let mut prev = self.points[0].to_pixel_point();
202                    prev.x += offset.x;
203                    prev.y += offset.y;
204
205                    let mut i = 1;
206                    while i < self.points.len() {
207                        let mut curr = self.points[i].to_pixel_point();
208                        curr.x += offset.x;
209                        curr.y += offset.y;
210
211                        if self.stroke_width <= 1 {
212                            ctx.draw_line(prev.x, prev.y, curr.x, curr.y, stroke)?;
213                        } else {
214                            ctx.draw_line_styled(
215                                prev.x,
216                                prev.y,
217                                curr.x,
218                                curr.y,
219                                StrokeStyle::new(stroke).with_width(self.stroke_width),
220                            )?;
221                        }
222                        prev = curr;
223                        i += 1;
224                    }
225
226                    if self.is_closed && self.points.len() >= 3 {
227                        let mut first = self.points[0].to_pixel_point();
228                        first.x += offset.x;
229                        first.y += offset.y;
230                        if self.stroke_width <= 1 {
231                            ctx.draw_line(prev.x, prev.y, first.x, first.y, stroke)?;
232                        } else {
233                            ctx.draw_line_styled(
234                                prev.x,
235                                prev.y,
236                                first.x,
237                                first.y,
238                                StrokeStyle::new(stroke).with_width(self.stroke_width),
239                            )?;
240                        }
241                    }
242                }
243            }
244        }
245        Ok(())
246    }
247}

249/// A composite Precise Draw Command Image (PDC vector asset).
250#[derive(Clone, Debug, PartialEq)]
251pub struct PdcImage<const MAX_COMMANDS: usize = 8, const MAX_POINTS_PER_CMD: usize = 16> {
252    pub viewbox: Rect,
253    pub commands: Vec<PdcCommand<MAX_POINTS_PER_CMD>, MAX_COMMANDS>,
254}
255
256impl<const MAX_COMMANDS: usize, const MAX_POINTS_PER_CMD: usize> Default
257    for PdcImage<MAX_COMMANDS, MAX_POINTS_PER_CMD>
258{
259    fn default() -> Self {
260        Self::new(Rect::new(0, 0, 0, 0))
261    }
262}
263
264impl<const MAX_COMMANDS: usize, const MAX_POINTS_PER_CMD: usize>
265    PdcImage<MAX_COMMANDS, MAX_POINTS_PER_CMD>
266{
267    pub const fn new(viewbox: Rect) -> Self {
268        Self {
269            viewbox,
270            commands: Vec::new(),
271        }
272    }
273
274    pub fn push_command(&mut self, cmd: PdcCommand<MAX_POINTS_PER_CMD>) -> Result<(), PdcError> {
275        self.commands.push(cmd).map_err(|_| PdcError)
276    }
277
278    /// Renders the entire vector image at the given target origin.
279    pub fn draw<D, C>(&self, ctx: &mut RenderCtx<'_, D, C>, origin: Point) -> Result<(), D::Error>
280    where
281        D: DrawTarget<Color = Rgb565>,
282        C: crate::render::Compositor<D>,
283    {
284        for cmd in &self.commands {
285            cmd.render(ctx, origin)?;
286        }
287        Ok(())
288    }
289}

291/// Pebble-inspired PDCS (Precise Draw Command Sequence) vector animation reel.
292///
293/// Cycles through keyframed vector frames with per-frame durations, supporting
294/// looping, pausing, rewinding, and zero-allocation frame rendering.
295#[derive(Clone, Debug)]
296pub struct PdcsReel<
297    const MAX_FRAMES: usize = 8,
298    const MAX_CMDS: usize = 8,
299    const MAX_POINTS: usize = 16,
300> {
301    pub frames: Vec<PdcImage<MAX_CMDS, MAX_POINTS>, MAX_FRAMES>,
302    pub durations_ms: [u32; MAX_FRAMES],
303    pub loop_count: u32,
304    pub completed_loops: u32,
305    pub current_frame: usize,
306    pub frame_elapsed_ms: u32,
307    pub is_playing: bool,
308}
309
310impl<const MAX_FRAMES: usize, const MAX_CMDS: usize, const MAX_POINTS: usize> Default
311    for PdcsReel<MAX_FRAMES, MAX_CMDS, MAX_POINTS>
312{
313    fn default() -> Self {
314        Self::new()
315    }
316}
317
318impl<const MAX_FRAMES: usize, const MAX_CMDS: usize, const MAX_POINTS: usize>
319    PdcsReel<MAX_FRAMES, MAX_CMDS, MAX_POINTS>
320{
321    pub const fn new() -> Self {
322        Self {
323            frames: Vec::new(),
324            durations_ms: [100; MAX_FRAMES],
325            loop_count: 0,
326            completed_loops: 0,
327            current_frame: 0,
328            frame_elapsed_ms: 0,
329            is_playing: true,
330        }
331    }
332
333    pub fn add_frame(
334        &mut self,
335        frame: PdcImage<MAX_CMDS, MAX_POINTS>,
336        duration_ms: u32,
337    ) -> Result<(), PdcError> {
338        let idx = self.frames.len();
339        if idx < MAX_FRAMES {
340            self.durations_ms[idx] = duration_ms.max(1);
341            self.frames.push(frame).map_err(|_| PdcError)
342        } else {
343            Err(PdcError)
344        }
345    }
346
347    pub fn play(&mut self) {
348        self.is_playing = true;
349    }
350
351    pub fn pause(&mut self) {
352        self.is_playing = false;
353    }
354
355    pub fn rewind(&mut self) {
356        self.current_frame = 0;
357        self.frame_elapsed_ms = 0;
358        self.completed_loops = 0;
359    }
360
361    pub fn is_finished(&self) -> bool {
362        self.loop_count > 0 && self.completed_loops >= self.loop_count
363    }
364
365    /// Advance sequence playback by `dt_ms` milliseconds.
366    pub fn update(&mut self, dt_ms: u32) {
367        if !self.is_playing || self.frames.is_empty() || self.is_finished() {
368            return;
369        }
370
371        self.frame_elapsed_ms += dt_ms;
372        let mut dur = self.durations_ms[self.current_frame];
373        while self.frame_elapsed_ms >= dur {
374            self.frame_elapsed_ms -= dur;
375            self.current_frame += 1;
376            if self.current_frame >= self.frames.len() {
377                self.current_frame = 0;
378                self.completed_loops += 1;
379                if self.is_finished() {
380                    self.is_playing = false;
381                    self.current_frame = self.frames.len() - 1;
382                    break;
383                }
384            }
385            dur = self.durations_ms[self.current_frame];
386        }
387    }
388
389    /// Current active vector frame image.
390    pub fn current_image(&self) -> Option<&PdcImage<MAX_CMDS, MAX_POINTS>> {
391        self.frames.get(self.current_frame)
392    }
393
394    /// Renders current animated vector frame to target.
395    pub fn draw<D, C>(&self, ctx: &mut RenderCtx<'_, D, C>, origin: Point) -> Result<(), D::Error>
396    where
397        D: DrawTarget<Color = Rgb565>,
398        C: crate::render::Compositor<D>,
399    {
400        if let Some(img) = self.current_image() {
401            img.draw(ctx, origin)?;
402        }
403        Ok(())
404    }
405}

413    #[test]
414    fn test_pdc_precise_points() {
415        let pt = PdcPrecisePoint::from_pixels(10, 20);
416        assert_eq!(pt.x_fixed, 80);
417        assert_eq!(pt.y_fixed, 160);
418        assert_eq!(pt.to_pixel_point(), Point::new(10, 20));
419
420        let sub_pt = PdcPrecisePoint::from_subpixels(84, 164);
421        assert_eq!(sub_pt.to_pixel_point(), Point::new(10, 20));
422        let (fx, fy) = sub_pt.to_f32_point();
423        assert!((fx - 10.5).abs() < 0.001);
424        assert!((fy - 20.5).abs() < 0.001);
425    }

428    fn test_pdc_image_render() {
429        let mut img = PdcImage::<4, 8>::new(Rect::new(0, 0, 20, 20));
430        let circle = PdcCommand::circle(Point::new(10, 10), 4, Some(Rgb565::RED), None, 1);

438    #[test]
439    fn test_pdcs_reel_playback() {
440        let mut reel = PdcsReel::<4, 2, 4>::new();
441        let frame1 = PdcImage::new(Rect::new(0, 0, 10, 10));
442        let frame2 = PdcImage::new(Rect::new(0, 0, 10, 10));
443
444        assert!(reel.add_frame(frame1, 50).is_ok());
445        assert!(reel.add_frame(frame2, 50).is_ok());
446        assert_eq!(reel.current_frame, 0);
447
448        reel.update(30);
449        assert_eq!(reel.current_frame, 0);
450
451        reel.update(25);
452        assert_eq!(reel.current_frame, 1);
453
454        reel.update(50);
455        // Loops back to frame 0
456        assert_eq!(reel.current_frame, 0);
457        assert_eq!(reel.completed_loops, 1);
458    }
