1//! 1-bit-per-pixel bitmaps and the composite icons built from them.
2//!
3//! Unlike [`ImageRef`](crate::image::ImageRef), the ink color is supplied at
4//! draw time, so one asset can be tinted per state. Composite icons stack
5//! several such parts at fixed offsets and toggle each part independently,
6//! which is how multi-part status glyphs stay a single addressable widget.
7
8use embedded_graphics_core::pixelcolor::Rgb565;
9
10/// A 1-bit-per-pixel bitmap: `bits` is row-major, MSB-first, each row padded to
11/// a byte boundary.
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub struct MonoBitmap<'a> {
14    pub width: u32,
15    pub height: u32,
16    pub bits: &'a [u8],
17}
18
19impl<'a> MonoBitmap<'a> {
20    pub const fn new(width: u32, height: u32, bits: &'a [u8]) -> Self {
21        Self {
22            width,
23            height,
24            bits,
25        }
26    }
27
28    /// Bytes per row, including the padding to the next byte boundary.
29    #[inline]
30    pub const fn stride(&self) -> u32 {
31        self.width.div_ceil(8)
32    }
33
34    /// Returns true when the source pixel at `(x, y)` is ink.
35    #[inline]
36    pub fn is_ink(&self, x: u32, y: u32) -> bool {
37        if x >= self.width || y >= self.height {
38            return false;
39        }
40        let idx = (y * self.stride() + x / 8) as usize;
41        match self.bits.get(idx) {
42            Some(byte) => byte & (0x80 >> (x % 8)) != 0,
43            None => false,
44        }
45    }
46}

48/// One layer of a [`WidgetKind::CompositeIcon`](crate::widgets::WidgetKind).
49///
50/// Offsets are in unscaled source pixels so a part keeps its position when the
51/// icon's `scale` changes.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub struct IconPart<'a> {
54    pub bitmap: MonoBitmap<'a>,
55    pub dx: i32,
56    pub dy: i32,
57    /// Drives state: a hidden part leaves its region untouched, so the icon
58    /// reads as incomplete rather than punching a hole in the backdrop.
59    pub visible: bool,
60    /// Per-part ink override; falls back to the icon's ink color.
61    pub tint: Option<Rgb565>,
62}
63
64impl<'a> IconPart<'a> {
65    pub const fn new(bitmap: MonoBitmap<'a>, dx: i32, dy: i32) -> Self {
66        Self {
67            bitmap,
68            dx,
69            dy,
70            visible: true,
71            tint: None,
72        }
73    }
74
75    pub const fn with_visible(mut self, visible: bool) -> Self {
76        self.visible = visible;
77        self
78    }
79
80    pub const fn with_tint(mut self, tint: Rgb565) -> Self {
81        self.tint = Some(tint);
82        self
83    }
84}

86/// Placement of a composite icon's parts inside its widget rect.
87#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
88pub enum IconAlign {
89    TopLeft,
90    #[default]
91    Center,
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn reads_msb_first_rows_with_byte_padding() {
100        // 9px wide => 2 bytes per row, second row starts at byte 2.
101        let bitmap = MonoBitmap::new(9, 2, &[0b1000_0000, 0b1000_0000, 0b0000_0001, 0b0000_0000]);
102        assert_eq!(bitmap.stride(), 2);
103        assert!(bitmap.is_ink(0, 0));
104        assert!(bitmap.is_ink(8, 0));
105        assert!(!bitmap.is_ink(1, 0));
106        assert!(bitmap.is_ink(7, 1));
107        assert!(!bitmap.is_ink(9, 0));
108        assert!(!bitmap.is_ink(0, 2));
109    }
