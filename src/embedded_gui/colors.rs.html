10/// Operations for tinting, blending, and lighting colors without floating-point math.
11///
12/// Adapted from `bevy_color::color_ops` (`Luminance`, `Mix`) for deterministic,
13/// fixed-point `#![no_std]` execution.
14pub trait ColorOps: Sized {
15    /// Darkens the color by `factor` where 0 leaves the color unchanged and 255 is pure black.
16    fn darker(self, factor: u8) -> Self;
17
18    /// Lightens the color by `factor` where 0 leaves the color unchanged and 255 is pure white.
19    fn lighter(self, factor: u8) -> Self;
20
21    /// Linearly interpolates between `self` and `other` where `factor == 0` is `self`
22    /// and `factor == 255` is `other`.
23    fn lerp(self, other: Self, factor: u8) -> Self;
24
25    /// Alpha blends `foreground` over `self` with alpha in `0..=255`.
26    fn blend(self, foreground: Self, alpha: u8) -> Self;
27
28    /// Calculates relative luminance in `0..=255` using standard Rec.601 integer coefficients.
29    fn luminance(&self) -> u8;
30}
31
32impl ColorOps for Rgb565 {
33    #[inline]
34    fn darker(self, factor: u8) -> Self {
35        self.lerp(Rgb565::new(0, 0, 0), factor)
36    }
37
38    #[inline]
39    fn lighter(self, factor: u8) -> Self {
40        self.lerp(Rgb565::new(31, 63, 31), factor)
41    }
42
43    #[inline]
44    fn lerp(self, other: Self, factor: u8) -> Self {
45        use embedded_graphics_core::pixelcolor::RgbColor;
46        let r0 = self.r() as u16;
47        let g0 = self.g() as u16;
48        let b0 = self.b() as u16;
49
50        let r1 = other.r() as u16;
51        let g1 = other.g() as u16;
52        let b1 = other.b() as u16;
53
54        let inv = 255 - factor as u16;
55        let f = factor as u16;
56
57        let r = ((r0 * inv + r1 * f + 127) / 255) as u8;
58        let g = ((g0 * inv + g1 * f + 127) / 255) as u8;
59        let b = ((b0 * inv + b1 * f + 127) / 255) as u8;
60
61        Rgb565::new(r.min(31), g.min(63), b.min(31))
62    }
63
64    #[inline]
65    fn blend(self, foreground: Self, alpha: u8) -> Self {
66        self.lerp(foreground, alpha)
67    }
68
69    #[inline]
70    fn luminance(&self) -> u8 {
71        use embedded_graphics_core::pixelcolor::RgbColor;
72        let r8 = (self.r() as u32 * 527 + 23) >> 6;
73        let g8 = (self.g() as u32 * 259 + 33) >> 6;
74        let b8 = (self.b() as u32 * 527 + 23) >> 6;
75        ((77 * r8 + 150 * g8 + 29 * b8) >> 8) as u8
76    }
77}

79/// Converts a 16-bit packed RGB565 raw value into [`Rgb565`].
80#[inline]
81pub const fn from_raw565(raw: u16) -> Rgb565 {
82    Rgb565::new(
83        ((raw >> 11) & 0x1F) as u8,
84        ((raw >> 5) & 0x3F) as u8,
85        (raw & 0x1F) as u8,
86    )
87}
88
89/// Converts 24-bit RGB888 components into [`Rgb565`].
90#[inline]
91pub const fn rgb888(r: u8, g: u8, b: u8) -> Rgb565 {
92    Rgb565::new(r >> 3, g >> 2, b >> 3)
93}
94
95/// Converts a 24-bit hex RGB value (e.g. `0xFF5733`) into [`Rgb565`].
96#[inline]
97pub const fn from_hex24(hex: u32) -> Rgb565 {
98    rgb888(
99        ((hex >> 16) & 0xFF) as u8,
100        ((hex >> 8) & 0xFF) as u8,
101        (hex & 0xFF) as u8,
102    )
103}

407/// Lookup a standard CSS or Tailwind color by name (case-insensitive, zero heap allocation).
408pub fn from_name(name: &str) -> Option<Rgb565> {
409    let mut i = 0;
410    while i < CSS_COLOR_TABLE.len() {
411        if CSS_COLOR_TABLE[i].0.eq_ignore_ascii_case(name) {
412            return Some(CSS_COLOR_TABLE[i].1);
413        }
414        i += 1;
415    }
416    if let Some(c) = css::from_name(name) {
417        return Some(c);
418    }
419    tailwind::from_name(name)
420}

454    #[test]
455    fn test_color_ops_rgb565() {
456        let white = Rgb565::new(31, 63, 31);
457        let black = Rgb565::new(0, 0, 0);
458
459        assert_eq!(white.darker(255), black);
460        assert_eq!(black.lighter(255), white);
461        assert_eq!(white.darker(0), white);
462        assert_eq!(black.lighter(0), black);
463
464        let mid = black.lerp(white, 128);
465        assert_eq!(mid.r(), 16);
466        assert_eq!(mid.g(), 32);
467        assert_eq!(mid.b(), 16);
468
469        assert_eq!(black.luminance(), 0);
470        assert_eq!(white.luminance(), 255);
471    }
