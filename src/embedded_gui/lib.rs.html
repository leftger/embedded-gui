102/// Canonical pixel color type for the active target configuration.
103///
104/// Defaults to [`embedded_graphics_core::pixelcolor::Rgb565`], but can be selected at compile time via the
105/// `color-rgb888` or `color-gray8` feature flags for zero-cost hardware specialization.
106#[cfg(feature = "color-rgb888")]
107pub type ActiveColor = embedded_graphics_core::pixelcolor::Rgb888;
108#[cfg(all(feature = "color-gray8", not(feature = "color-rgb888")))]
109pub type ActiveColor = embedded_graphics_core::pixelcolor::Gray8;
110#[cfg(not(any(feature = "color-rgb888", feature = "color-gray8")))]
111pub type ActiveColor = embedded_graphics_core::pixelcolor::Rgb565;
