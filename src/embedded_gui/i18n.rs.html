1//! Zero-allocation internationalization (i18n) and localization for `#![no_std]` embedded targets.
2//! Allows microcontrollers to store compact string translation tables in Flash and switch languages
3//! dynamically at runtime without heap allocation.
4
5#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
6pub struct LanguageId(pub u8);
7
8impl LanguageId {
9    pub const EN: Self = Self(0);
10    pub const ES: Self = Self(1);
11    pub const DE: Self = Self(2);
12    pub const FR: Self = Self(3);
13    pub const IT: Self = Self(4);
14    pub const JA: Self = Self(5);
15    pub const ZH: Self = Self(6);
16
17    pub const fn new(id: u8) -> Self {
18        Self(id)
19    }
20
21    pub const fn raw(self) -> u8 {
22        self.0
23    }
24}
25
26/// A single translation entry mapping a string key to localized strings across language indices.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub struct TranslationEntry<'a> {
29    pub key: &'a str,
30    pub translations: &'a [&'a str],
31}
32
33impl<'a> TranslationEntry<'a> {
34    pub const fn new(key: &'a str, translations: &'a [&'a str]) -> Self {
35        Self { key, translations }
36    }
37
38    #[inline]
39    pub fn get(&self, lang: LanguageId) -> Option<&'a str> {
40        let idx = lang.0 as usize;
41        if idx < self.translations.len() {
42            Some(self.translations[idx])
43        } else {
44            None
45        }
46    }
47}
48
49/// A zero-allocation static translation table stored in ROM/Flash.
50#[derive(Clone, Copy, Debug)]
51pub struct TranslationTable<'a> {
52    pub entries: &'a [TranslationEntry<'a>],
53    pub active_lang: LanguageId,
54    pub fallback_lang: LanguageId,
55}
56
57impl<'a> TranslationTable<'a> {
58    pub const fn new(entries: &'a [TranslationEntry<'a>]) -> Self {
59        Self {
60            entries,
61            active_lang: LanguageId::EN,
62            fallback_lang: LanguageId::EN,
63        }
64    }
65
66    pub const fn with_active_lang(mut self, lang: LanguageId) -> Self {
67        self.active_lang = lang;
68        self
69    }
70
71    pub fn set_language(&mut self, lang: LanguageId) {
72        self.active_lang = lang;
73    }
74
75    /// Looks up a localized string by key using the active language, falling back to fallback_lang or the raw key.
76    pub fn translate(&self, key: &'a str) -> &'a str {
77        if let Some(entry) = self.entries.iter().find(|e| e.key == key) {
78            if let Some(s) = entry.get(self.active_lang) {
79                return s;
80            }
81            if let Some(s) = entry.get(self.fallback_lang) {
82                return s;
83            }
84        }
85        key
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    static ENTRIES: [TranslationEntry<'static>; 3] = [
94        TranslationEntry::new("btn_start", &["Start", "Iniciar", "Starten"]),
95        TranslationEntry::new("btn_stop", &["Stop", "Detener", "Stoppen"]),
96        TranslationEntry::new("status_ready", &["Ready", "Listo", "Bereit"]),
97    ];
98
99    #[test]
100    fn test_translation_table_lookup() {
101        let mut tr = TranslationTable::new(&ENTRIES);
102
103        assert_eq!(tr.translate("btn_start"), "Start");
104        assert_eq!(tr.translate("btn_stop"), "Stop");
105
106        // Switch to Spanish
107        tr.set_language(LanguageId::ES);
108        assert_eq!(tr.translate("btn_start"), "Iniciar");
109        assert_eq!(tr.translate("btn_stop"), "Detener");
110        assert_eq!(tr.translate("status_ready"), "Listo");
111
112        // Switch to German
113        tr.set_language(LanguageId::DE);
114        assert_eq!(tr.translate("btn_start"), "Starten");
115        assert_eq!(tr.translate("status_ready"), "Bereit");
116
117        // Fallback for missing key
118        assert_eq!(tr.translate("unknown_key"), "unknown_key");
119    }
