29/// High-level combined 3D scene + 2D GUI overlay rendering pipeline.
30pub struct Gui3dPipeline<
31    'a,
32    const MAX_NODES: usize,
33    const MAX_HANDLERS: usize,
34    const MAX_ACTIVE: usize,
35> {
36    /// 3D rendering engine instance.
37    pub engine: K3dengine,
38    /// 2D GUI context instance.
39    pub gui: GuiContext<'a, MAX_NODES, MAX_HANDLERS, MAX_ACTIVE>,
40    zbuffer: &'a mut [u32],
41    commands: CommandBuffer<64>,
42}
43
44impl<'a, const MAX_NODES: usize, const MAX_HANDLERS: usize, const MAX_ACTIVE: usize>
45    Gui3dPipeline<'a, MAX_NODES, MAX_HANDLERS, MAX_ACTIVE>
46{
47    /// Create a new pipeline given screen dimensions and a Z-buffer slice.
48    pub fn new(width: usize, height: usize, zbuffer: &'a mut [u32]) -> Self {
49        Self {
50            engine: K3dengine::new(width as u16, height as u16),
51            gui: GuiContext::new(Rect::new(0, 0, width as u32, height as u32)),
52            zbuffer,
53            commands: CommandBuffer::new(),
54        }
55    }
56
57    /// Render a frame containing 3D meshes followed by dirty 2D GUI widgets onto `target`.
58    pub fn render_frame<'m, D>(
59        &mut self,
60        target: &mut D,
61        meshes: impl IntoIterator<Item = &'m K3dMesh<'m>>,
62    ) -> Result<(), D::Error>
63    where
64        D: DrawTarget<Color = Rgb565> + OriginDimensions,
65        D::Error: Debug,
66    {
67        self.commands.clear();
68        self.engine.record(meshes, &mut self.commands, None).ok();
69
70        let vp = self.gui.viewport();
71        let mut frame = FrameCtx {
72            zbuffer: self.zbuffer,
73            width: vp.w as usize,
74            height: vp.h as usize,
75        };
76        self.engine
77            .execute(target, &mut frame, &self.commands, None)
78            .ok();
79
80        self.gui.render(target)?;
81        self.gui.clear_dirty();
82
83        Ok(())
84    }
85}

102/// A single mesh framed inside a GUI rect: the 3D counterpart of an image
103/// widget, used for spinning logos and other decorative geometry.
104///
105/// The camera looks at the origin from `camera_distance` along +Z, so a mesh
106/// centered on the origin stays framed no matter how it is rotated.
107pub struct MeshPanel<'a> {
108    pub geometry: Geometry<'a>,
109    pub color: Rgb565,
110    pub shading: MeshShading,
111    pub scale: f32,
112    /// Roll, pitch, yaw in radians. Animate these for spin or coin-flip motion.
113    pub attitude: (f32, f32, f32),
114    pub camera_distance: f32,
115    /// Vertical field of view in radians.
116    pub fov: f32,
117    pub light_dir: [f32; 3],
118}
119
120impl<'a> MeshPanel<'a> {
121    pub fn new(geometry: Geometry<'a>, color: Rgb565) -> Self {
122        Self {
123            geometry,
124            color,
125            shading: MeshShading::Solid,
126            scale: 1.0,
127            attitude: (0.0, 0.0, 0.0),
128            camera_distance: 4.0,
129            fov: core::f32::consts::FRAC_PI_2,
130            light_dir: [0.0, 0.0, -1.0],
131        }
132    }
133}

135/// Clips and translates draws into a sub-rect, so the 3D rasterizer can render
136/// at panel-local coordinates while writing into the shared framebuffer.
137struct OffsetTarget<'d, D> {
138    inner: &'d mut D,
139    origin: Point,
140    size: Size,
141}
142
143impl<D> OriginDimensions for OffsetTarget<'_, D> {
144    fn size(&self) -> Size {
145        self.size
146    }
147}
148
149impl<D> DrawTarget for OffsetTarget<'_, D>
150where
151    D: DrawTarget<Color = Rgb565>,
152{
153    type Color = Rgb565;
154    type Error = D::Error;
155
156    fn draw_iter<I>(&mut self, pixels: I) -> Result<(), Self::Error>
157    where
158        I: IntoIterator<Item = Pixel<Self::Color>>,
159    {
160        let w = self.size.width as i32;
161        let h = self.size.height as i32;
162        let origin = self.origin;
163        self.inner.draw_iter(pixels.into_iter().filter_map(
164            move |Pixel(point, color)| match point {
165                p if p.x >= 0 && p.y >= 0 && p.x < w && p.y < h => {
166                    Some(Pixel(Point::new(p.x + origin.x, p.y + origin.y), color))
167                }
168                _ => None,
169            },
170        ))
171    }
172
173    fn fill_solid(&mut self, area: &Rectangle, color: Self::Color) -> Result<(), Self::Error> {
174        let clipped = area.intersection(&Rectangle::new(Point::zero(), self.size));
175        if clipped.size.width == 0 || clipped.size.height == 0 {
176            return Ok(());
177        }
178        self.inner.fill_solid(
179            &Rectangle::new(
180                Point::new(
181                    clipped.top_left.x + self.origin.x,
182                    clipped.top_left.y + self.origin.y,
183                ),
184                clipped.size,
185            ),
186            color,
187        )
188    }
189}

191/// Renders `panel` into `rect` of `target`.
192///
193/// `zbuffer` must hold at least `rect.w * rect.h` entries; it is cleared by the
194/// engine on each call, so one scratch buffer can be shared by every panel.
195pub fn render_mesh_panel<D>(
196    target: &mut D,
197    rect: Rect,
198    panel: &MeshPanel<'_>,
199    zbuffer: &mut [u32],
200) -> Result<(), MeshPanelError>
201where
202    D: DrawTarget<Color = Rgb565>,
203    D::Error: Debug,
204{
205    if rect.is_empty() {
206        return Ok(());
207    }
208    let pixels = rect.w as usize * rect.h as usize;
209    if zbuffer.len() < pixels {
210        return Err(MeshPanelError::ZBufferTooSmall {
211            needed: pixels,
212            got: zbuffer.len(),
213        });
214    }
215
216    let mut engine = K3dengine::new(rect.w as u16, rect.h as u16);
217    engine
218        .camera
219        .set_position(nalgebra::Point3::new(0.0, 0.0, panel.camera_distance));
220    engine.camera.set_target(nalgebra::Point3::origin());
221    engine.camera.set_fovy(panel.fov);
222
223    let mut mesh = K3dMesh::new(panel.geometry);
224    mesh.set_color(panel.color);
225    mesh.set_scale(panel.scale);
226    let (roll, pitch, yaw) = panel.attitude;
227    mesh.set_attitude(roll, pitch, yaw);
228    mesh.set_render_mode(match panel.shading {
229        MeshShading::Points => RenderMode::Points,
230        MeshShading::Lines => RenderMode::Lines,
231        MeshShading::Solid => RenderMode::Solid,
232        MeshShading::Lit => RenderMode::SolidLightDir(nalgebra::Vector3::new(
233            panel.light_dir[0],
234            panel.light_dir[1],
235            panel.light_dir[2],
236        )),
237    });
238
239    let mut commands = CommandBuffer::<512>::new();
240    engine
241        .record([&mesh], &mut commands, None)
242        .map_err(|_| MeshPanelError::RecordFailed)?;
243
244    let mut frame = FrameCtx {
245        zbuffer: &mut zbuffer[..pixels],
246        width: rect.w as usize,
247        height: rect.h as usize,
248    };
249    let mut view = OffsetTarget {
250        inner: target,
251        origin: Point::new(rect.x, rect.y),
252        size: Size::new(rect.w, rect.h),
253    };
254    engine
255        .execute(&mut view, &mut frame, &commands, None)
256        .map_err(|_| MeshPanelError::ExecuteFailed)?;
257    Ok(())
258}

260/// Why a [`render_mesh_panel`] call could not draw.

262pub enum MeshPanelError {

264        needed: usize,
265        got: usize,
266    },

268    RecordFailed,
269    ExecuteFailed,
270}

272/// Extension trait for spatial 3D world anchors on `GuiContext`.
273pub trait WorldAnchorExt {
274    /// Anchor a widget to a 3D world coordinate using camera projection.
275    fn anchor_widget_to_world(
276        &mut self,
277        widget_id: WidgetId,
278        world_pos: Point3<f32>,
279        engine: &K3dengine,
280        offset: Point,
281    ) -> bool;
282}
283
284impl<'a, const MAX_NODES: usize, const MAX_HANDLERS: usize, const MAX_ACTIVE: usize> WorldAnchorExt
285    for GuiContext<'a, MAX_NODES, MAX_HANDLERS, MAX_ACTIVE>
286{
287    fn anchor_widget_to_world(
288        &mut self,
289        widget_id: WidgetId,
290        world_pos: Point3<f32>,
291        engine: &K3dengine,
292        offset: Point,
293    ) -> bool {
294        if let Some(screen_pt) = engine.project_point(world_pos) {
295            let pos = Point::new(screen_pt.x + offset.x, screen_pt.y + offset.y);
296            if let Some(rect) = self.absolute_rect(widget_id) {
297                let new_rect = Rect::new(pos.x, pos.y, rect.w, rect.h);
298                if let Some(node) = self.node_mut(widget_id) {
299                    node.rect = new_rect;
300                    let _ = self.dirty.add(new_rect);
301                    return true;
302                }
303            }
304        }
305        false
306    }
307}

309/// Result of dispatching an input event in a 3D+GUI application.
310#[derive(Clone, Copy, Debug)]
311pub enum InputResult {
312    /// Handled by 2D GUI.
313    GuiHandled(UiEvent),
314    /// Unhandled by GUI; converted into a 3D pick ray.
315    ScenePick(Ray),
316    /// Event ignored / unhandled.
317    Ignored,
318}
319
320/// Router for dispatching pointer events to 2D GUI first, then 3D picking rays.
321pub fn dispatch_pointer_input<
322    'a,
323    const MAX_NODES: usize,
324    const MAX_HANDLERS: usize,
325    const MAX_ACTIVE: usize,
326>(
327    gui: &mut GuiContext<'a, MAX_NODES, MAX_HANDLERS, MAX_ACTIVE>,
328    engine: &K3dengine,
329    x: i32,
330    y: i32,
331    state: PointerState,
332    button: PointerButton,
333) -> InputResult {
334    let evt = InputEvent::Pointer {
335        x,
336        y,
337        state,
338        button,
339    };
340    let _ = gui.handle_input(evt);
341    if let Some(ui_evt) = gui.pop_event() {
342        InputResult::GuiHandled(ui_evt)
343    } else if state == PointerState::Pressed {
344        let vp = gui.viewport();
345        let ray = Ray::from_screen_point(
346            Point::new(x, y),
347            &engine.camera,
348            vp.w as usize,
349            vp.h as usize,
350        );
351        InputResult::ScenePick(ray)
352    } else {
353        InputResult::Ignored
354    }
355}

357struct SliceBackend<'b>(&'b mut [Rgb565]);
358
359impl FrameBufferBackend for SliceBackend<'_> {
360    type Color = Rgb565;
361    fn set(&mut self, index: usize, color: Rgb565) {
362        self.0[index] = color;
363    }
364    fn get(&self, index: usize) -> Rgb565 {
365        self.0[index]

367    fn nr_elements(&self) -> usize {

369    }

372/// Extension trait for rendering GUI trees directly into textures for 3D meshes.
373#[allow(clippy::result_unit_err)]
374pub trait RenderToTextureExt {
375    /// Render the GUI context into a slice buffer to be used as a 3D texture.
376    fn render_to_texture(
377        &mut self,
378        buffer: &mut [Rgb565],
379        width: usize,
380        height: usize,
381    ) -> Result<(), ()>;
382}
383
384impl<'a, const MAX_NODES: usize, const MAX_HANDLERS: usize, const MAX_ACTIVE: usize>
385    RenderToTextureExt for GuiContext<'a, MAX_NODES, MAX_HANDLERS, MAX_ACTIVE>
386{
387    #[allow(clippy::result_unit_err)]
388    fn render_to_texture(
389        &mut self,
390        buffer: &mut [Rgb565],
391        width: usize,
392        height: usize,
393    ) -> Result<(), ()> {
394        if buffer.len() != width * height {
395            return Err(());
396        }
397        let mut fb = FrameBuf::new(SliceBackend(buffer), width, height);
398        self.render(&mut fb).map_err(|_| ())?;
399        Ok(())
400    }
