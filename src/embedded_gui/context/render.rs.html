36    pub fn render_dirty<D>(&self, target: &mut D) -> Result<(), D::Error>
37    where
38        D: embedded_graphics_core::draw_target::DrawTarget<Color = Rgb565>,
39    {
40        let slice = self.dirty.as_slice();
41        if slice.is_empty() {
42            return Ok(());
43        }
44
45        if slice.len() == 1 {
46            let mut ctx = RenderCtx::with_dirty(target, self.viewport, slice[0]);
47            ctx.set_quality(self.render_quality);
48            return self.render_into(&mut ctx, 0, 0, 255);
49        }
50
51        if let Some(bound) = self.dirty.bounding_rect() {
52            let area_bound = bound.w as u64 * bound.h as u64;
53            let sum_area: u64 = slice.iter().map(|r| r.w as u64 * r.h as u64).sum();
54            // If bounding rect area is close to sum of individual areas (overlap or cluster),
55            // render bounding rect in 1 pass to avoid multiple tree sweeps
56            if area_bound <= sum_area + (sum_area / 2) {
57                let mut ctx = RenderCtx::with_dirty(target, self.viewport, bound);
58                ctx.set_quality(self.render_quality);
59                return self.render_into(&mut ctx, 0, 0, 255);
60            }
61        }
62
63        for dirty in slice {
64            let mut ctx = RenderCtx::with_dirty(target, self.viewport, *dirty);
65            ctx.set_quality(self.render_quality);
66            self.render_into(&mut ctx, 0, 0, 255)?;
67        }
68        Ok(())
69    }

71    /// Renders dirty regions to a hardware display controller implementing [`crate::render::WindowedDrawTarget`].
72    /// Sets the physical controller's column/row window bounds (`set_window`) before rendering,
73    /// enabling hardware SPI/DMA transfers exclusively to the dirty sub-window.
74    pub fn render_dirty_windowed<D>(&self, target: &mut D) -> Result<(), D::Error>
75    where
76        D: embedded_graphics_core::draw_target::DrawTarget<Color = Rgb565>
77            + crate::render::WindowedDrawTarget,
78    {
79        let slice = self.dirty.as_slice();
80        if slice.is_empty() {
81            return Ok(());
82        }
83
84        for dirty in slice {
85            let eg_rect = embedded_graphics_core::primitives::Rectangle::new(
86                embedded_graphics_core::geometry::Point::new(dirty.x, dirty.y),
87                embedded_graphics_core::geometry::Size::new(dirty.w, dirty.h),
88            );
89            target.set_window(&eg_rect)?;
90            let mut ctx = RenderCtx::with_dirty(target, self.viewport, *dirty);
91            ctx.set_quality(self.render_quality);
92            self.render_into(&mut ctx, 0, 0, 255)?;
93        }
94        Ok(())
95    }

97    /// Renders dirty regions into a small caller-provided scratch pixel buffer, then
98    /// contiguously flushes each region to the display target.
99    ///
100    /// This eliminates visual flicker and dramatically speeds up partial updates on
101    /// SPI/I2C displays without requiring a full RAM framebuffer.
102    pub fn render_dirty_buffered<D>(
103        &self,
104        target: &mut D,
105        scratch_buffer: &mut [Rgb565],
106    ) -> Result<(), D::Error>
107    where
108        D: embedded_graphics_core::draw_target::DrawTarget<Color = Rgb565>,
109    {
110        let slice = self.dirty.as_slice();
111        if slice.is_empty() || scratch_buffer.is_empty() {
112            return Ok(());
113        }
114
115        for &dirty in slice {
116            if dirty.is_empty() {
117                continue;
118            }
119            let total_pixels = (dirty.w as usize).saturating_mul(dirty.h as usize);
120            if total_pixels <= scratch_buffer.len() {
121                // Entire dirty rect fits in the scratch buffer
122                let mut fb =
123                    crate::framebuffer::FramebufferSlice::new(scratch_buffer, dirty.w, dirty.h);
124                let bg = self.theme.panel.background.unwrap_or(Rgb565::new(0, 0, 0));
125                fb.clear_color(bg);
126                let mut ctx = RenderCtx::new(&mut fb, Rect::new(0, 0, dirty.w, dirty.h));
127                ctx.set_quality(self.render_quality);
128                let _ = self.render_into(&mut ctx, -dirty.x, -dirty.y, 255);
129
130                let area = embedded_graphics_core::primitives::Rectangle::new(
131                    embedded_graphics_core::geometry::Point::new(dirty.x, dirty.y),
132                    embedded_graphics_core::geometry::Size::new(dirty.w, dirty.h),
133                );
134                target.fill_contiguous(&area, scratch_buffer[..total_pixels].iter().copied())?;
135            } else if scratch_buffer.len() >= dirty.w as usize {
136                // Buffer can fit at least one row: render in vertical bands
137                let band_height = (scratch_buffer.len() / dirty.w as usize) as u32;
138                let mut y = dirty.y;
139                let bottom = dirty.bottom();
140                while y < bottom {
141                    let h = band_height.min((bottom - y) as u32);
142                    let band_pixels = (dirty.w * h) as usize;
143                    let mut fb =
144                        crate::framebuffer::FramebufferSlice::new(scratch_buffer, dirty.w, h);
145                    let bg = self.theme.panel.background.unwrap_or(Rgb565::new(0, 0, 0));
146                    fb.clear_color(bg);
147                    let mut ctx = RenderCtx::new(&mut fb, Rect::new(0, 0, dirty.w, h));
148                    ctx.set_quality(self.render_quality);
149                    let _ = self.render_into(&mut ctx, -dirty.x, -y, 255);
150
151                    let area = embedded_graphics_core::primitives::Rectangle::new(
152                        embedded_graphics_core::geometry::Point::new(dirty.x, y),
153                        embedded_graphics_core::geometry::Size::new(dirty.w, h),
154                    );
155                    target.fill_contiguous(&area, scratch_buffer[..band_pixels].iter().copied())?;
156                    y += h as i32;
157                }
158            } else {
159                // Micro-scratch buffer (smaller than 1 scanline): chunk sub-row rectangles
160                let chunk_w = scratch_buffer.len() as u32;
161                let mut y = dirty.y;
162                let bottom = dirty.bottom();
163                while y < bottom {
164                    let mut x = dirty.x;
165                    let right = dirty.right();
166                    while x < right {
167                        let w = chunk_w.min((right - x) as u32);
168                        let chunk_pixels = w as usize;
169                        let mut fb =
170                            crate::framebuffer::FramebufferSlice::new(scratch_buffer, w, 1);
171                        let bg = self.theme.panel.background.unwrap_or(Rgb565::new(0, 0, 0));
172                        fb.clear_color(bg);
173                        let mut ctx = RenderCtx::new(&mut fb, Rect::new(0, 0, w, 1));
174                        ctx.set_quality(self.render_quality);
175                        let _ = self.render_into(&mut ctx, -x, -y, 255);
176
177                        let area = embedded_graphics_core::primitives::Rectangle::new(
178                            embedded_graphics_core::geometry::Point::new(x, y),
179                            embedded_graphics_core::geometry::Size::new(w, 1),
180                        );
181                        target.fill_contiguous(
182                            &area,
183                            scratch_buffer[..chunk_pixels].iter().copied(),
184                        )?;
185                        x += w as i32;
186                    }
187                    y += 1;
188                }
189            }
190        }
191        Ok(())
192    }

221    pub fn render_with_offset_opacity_and_clip<D>(
222        &self,
223        target: &mut D,
224        offset_x: i32,
225        offset_y: i32,
226        opacity: u8,
227        clip: Rect,
228    ) -> Result<(), D::Error>
229    where

235        ctx.set_clip(old_clip.intersection(clip));
236        self.render_into(&mut ctx, offset_x, offset_y, opacity)
237    }

239    pub(crate) fn render_into<D, C>(
240        &self,
241        ctx: &mut RenderCtx<'_, D, C>,
242        offset_x: i32,
243        offset_y: i32,
244        opacity: u8,
245    ) -> Result<(), D::Error>
246    where
247        D: embedded_graphics_core::draw_target::DrawTarget<Color = Rgb565>,
248        C: crate::render::Compositor<D>,
249    {
250        for node in &self.widgets {
251            if !self.effective_visible(node.id) {
252                continue;
253            }
254            let Some(base_rect) = self.absolute_rect(node.id) else {
255                continue;
256            };
257            let rect = Rect::new(
258                base_rect.x + offset_x,
259                base_rect.y + offset_y,
260                base_rect.w,
261                base_rect.h,
262            );
263            let base_clip = self.inherited_clip(node.id).unwrap_or(self.viewport);
264            let clip = Rect::new(
265                base_clip.x + offset_x,
266                base_clip.y + offset_y,
267                base_clip.w,
268                base_clip.h,
269            );
270            if rect.intersection(clip).is_empty() {
271                continue;
272            }
273            let old_clip = ctx.clip();
274            ctx.set_clip(old_clip.intersection(clip));
275            let state = if self.pressed.is_some_and(|pressed| pressed.id == node.id) {
276                VisualState::Pressed
277            } else if Some(node.id) == self.focus {
278                VisualState::Focused
279            } else if !self.effective_enabled(node.id) {
280                VisualState::Disabled
281            } else {
282                VisualState::Normal
283            };
284            let mut render_node = *node;
285            let class_style = node.style_class.and_then(|class| {
286                self.class_styles
287                    .iter()
288                    .find(|(id, _)| *id == class)
289                    .map(|(_, style)| *style)
290            });
291            let resolve_state_style = |vs: VisualState| {
292                class_style
293                    .map(|style| style.resolve(vs))
294                    .unwrap_or_else(|| render_node.style.resolve(vs))
295            };
296            let active_style = if let Some((from, to, t)) = self.state_transition_progress(node.id)
297            {
298                lerp_style(resolve_state_style(from), resolve_state_style(to), t)
299            } else {
300                resolve_state_style(state)
301            };
302            render_node.style = render_node.style.with_state_override(state, active_style);
303            if opacity < 255 {
304                let apply = |v: u8| -> u8 { ((v as u16 * opacity as u16) / 255) as u8 };
305                render_node.style.normal.opacity = apply(render_node.style.normal.opacity);
306                render_node.style.focused.opacity = apply(render_node.style.focused.opacity);
307                render_node.style.pressed.opacity = apply(render_node.style.pressed.opacity);
308                render_node.style.disabled.opacity = apply(render_node.style.disabled.opacity);
309            }
310            render_node.render_at(ctx, rect, state)?;
311            ctx.set_clip(old_clip);
312        }
313        Ok(())
314    }
