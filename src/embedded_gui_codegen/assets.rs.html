1//! Conversion of source art into the forms `embedded-gui` draws.
2//!
3//! Both the `include_gui!` macro and the Studio preview call into here, which
4//! is what keeps a design's preview and its firmware output byte-identical.
5//! Nothing in this module decodes container formats: callers hand over raw
6//! RGBA8 or file text, so the crate stays dependency-free.
7
8use std::collections::BTreeMap;
9use std::fmt;

11#[derive(Clone, Debug, PartialEq, Eq)]
12pub enum AssetError {
13    Bdf(String),
14    Obj(String),
15    Stl(String),
16    UnsupportedMesh(String),
17    Empty(&'static str),
18}
19
20impl fmt::Display for AssetError {
21    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
22        match self {
23            AssetError::Bdf(msg) => write!(f, "invalid BDF font: {msg}"),
24            AssetError::Obj(msg) => write!(f, "invalid OBJ mesh: {msg}"),
25            AssetError::Stl(msg) => write!(f, "invalid STL mesh: {msg}"),
26            AssetError::UnsupportedMesh(name) => {
27                write!(f, "'{name}' is not a supported mesh format (.obj or .stl)")
28            }
29            AssetError::Empty(what) => write!(f, "{what} contained no usable data"),
30        }
31    }
32}

36/// A 1-bit-per-pixel bitmap, row-major and MSB-first, each row padded to a byte
37/// boundary. Matches `embedded_gui::MonoBitmap`.
38#[derive(Clone, Debug, PartialEq, Eq)]
39pub struct MonoBitmapData {
40    pub width: u32,
41    pub height: u32,
42    pub bits: Vec<u8>,
43}
44
45impl MonoBitmapData {
46    pub fn stride(&self) -> u32 {
47        self.width.div_ceil(8)
48    }
49}

51/// Threshold RGBA8 pixels into ink/paper.
52///
53/// A pixel is ink when it is opaque enough and its luminance falls on the
54/// requested side of `threshold`. Art is usually black-on-white, so `invert`
55/// defaults off in the KDL layer and flips for white-on-black sources.
56pub fn mono_from_rgba(
57    width: u32,
58    height: u32,
59    rgba: &[u8],
60    threshold: u8,
61    invert: bool,
62) -> MonoBitmapData {
63    let stride = width.div_ceil(8);
64    let mut bits = vec![0u8; (stride * height) as usize];
65    for y in 0..height {
66        for x in 0..width {
67            let idx = ((y * width + x) * 4) as usize;
68            let Some(px) = rgba.get(idx..idx + 4) else {
69                continue;
70            };
71            let alpha = px[3];
72            let luminance =
73                ((u16::from(px[0]) * 77 + u16::from(px[1]) * 150 + u16::from(px[2]) * 29) >> 8)
74                    as u8;
75            let dark = luminance < threshold;
76            let ink = alpha >= 128 && (dark != invert);
77            if ink {
78                let byte = (y * stride + x / 8) as usize;
79                bits[byte] |= 0x80 >> (x % 8);
80            }
81        }
82    }
83    MonoBitmapData {
84        width,
85        height,
86        bits,
87    }
88}

90/// A monospaced bitmap font, laid out as `embedded_gui::BitmapFont` expects:
91/// glyphs stored back to back over the contiguous range starting at
92/// `first_char`, each glyph `height * bytes_per_row` bytes.
93#[derive(Clone, Debug, PartialEq, Eq)]
94pub struct BitmapFontData {
95    pub width: u8,
96    pub height: u8,
97    pub advance: u8,
98    pub line_height: u8,
99    pub first_char: u8,
100    pub bytes_per_row: u8,
101    pub glyphs: Vec<u8>,
102}

104struct BdfGlyph {
105    encoding: u32,
106    /// Glyph bounding box: width, height, x offset, y offset from baseline.
107    bbx: (i32, i32, i32, i32),
108    dwidth: i32,
109    rows: Vec<Vec<u8>>,
110}

112/// Parses a BDF font into a fixed-cell bitmap font.
113///
114/// Glyphs are composited into the font-wide bounding box so every cell shares
115/// an origin, which is what makes the result safe to index by character code.
116/// `chars`, when given, restricts the range that gets embedded: a shot counter
117/// only needs digits, and dropping the rest saves flash.
118pub fn parse_bdf(source: &str, chars: Option<&str>) -> Result<BitmapFontData, AssetError> {
119    let mut font_bbox: Option<(i32, i32, i32, i32)> = None;
120    let mut glyphs: BTreeMap<u32, BdfGlyph> = BTreeMap::new();
121
122    let mut current: Option<BdfGlyph> = None;
123    let mut in_bitmap = false;
124
125    for line in source.lines() {
126        let line = line.trim();
127        if line.is_empty() {
128            continue;
129        }
130        let mut parts = line.split_whitespace();
131        let keyword = parts.next().unwrap_or("");
132
133        if in_bitmap {
134            if keyword == "ENDCHAR" {
135                in_bitmap = false;
136                if let Some(glyph) = current.take() {
137                    glyphs.insert(glyph.encoding, glyph);
138                }
139                continue;
140            }
141            let bytes =
142                hex_row(line).ok_or_else(|| AssetError::Bdf(format!("bad BITMAP row '{line}'")))?;
143            if let Some(glyph) = current.as_mut() {
144                glyph.rows.push(bytes);
145            }
146            continue;
147        }
148
149        match keyword {
150            "FONTBOUNDINGBOX" => {
151                let v = read_ints(parts, 4)
152                    .ok_or_else(|| AssetError::Bdf("malformed FONTBOUNDINGBOX".into()))?;
153                font_bbox = Some((v[0], v[1], v[2], v[3]));
154            }
155            "STARTCHAR" => {
156                current = Some(BdfGlyph {
157                    encoding: 0,
158                    bbx: (0, 0, 0, 0),
159                    dwidth: 0,
160                    rows: Vec::new(),
161                });
162            }
163            "ENCODING" => {
164                if let (Some(glyph), Some(code)) =
165                    (current.as_mut(), parts.next().and_then(|v| v.parse().ok()))
166                {
167                    glyph.encoding = code;
168                }
169            }
170            "DWIDTH" => {
171                if let (Some(glyph), Some(w)) = (
172                    current.as_mut(),
173                    parts.next().and_then(|v| v.parse::<i32>().ok()),
174                ) {
175                    glyph.dwidth = w;
176                }
177            }
178            "BBX" => {
179                let v =
180                    read_ints(parts, 4).ok_or_else(|| AssetError::Bdf("malformed BBX".into()))?;
181                if let Some(glyph) = current.as_mut() {
182                    glyph.bbx = (v[0], v[1], v[2], v[3]);
183                }
184            }
185            "BITMAP" => in_bitmap = true,
186            _ => {}
187        }
188    }
189
190    let (fbb_w, fbb_h, ..) =
191        font_bbox.ok_or_else(|| AssetError::Bdf("missing FONTBOUNDINGBOX".into()))?;
192    if fbb_w <= 0 || fbb_h <= 0 {
193        return Err(AssetError::Bdf("font bounding box has no area".into()));
194    }
195
196    let wanted: Option<Vec<u32>> = chars.map(|s| s.chars().map(|c| c as u32).collect());
197    let selected: Vec<u32> = glyphs
198        .keys()
199        .copied()
200        .filter(|code| *code <= 255 && wanted.as_ref().is_none_or(|list| list.contains(code)))
201        .collect();
202    if selected.is_empty() {
203        return Err(AssetError::Empty("BDF font"));
204    }
205
206    // Size the cell to the glyphs actually being embedded rather than to
207    // FONTBOUNDINGBOX. Display fonts reserve room for accents and descenders no
208    // digit uses, and on a 96px-wide panel that padding is the difference
209    // between two digits fitting and one.
210    let mut left = i32::MAX;
211    let mut right = i32::MIN;
212    let mut bottom = i32::MAX;
213    let mut top = i32::MIN;
214    for code in &selected {
215        let (gw, gh, gx, gy) = glyphs[code].bbx;
216        left = left.min(gx);
217        right = right.max(gx + gw);
218        bottom = bottom.min(gy);
219        top = top.max(gy + gh);
220    }
221    let cell_w = right - left;
222    let cell_h = top - bottom;
223    if cell_w <= 0 || cell_h <= 0 {
224        return Err(AssetError::Empty("BDF font"));
225    }
226    if cell_w > 255 || cell_h > 255 {
227        return Err(AssetError::Bdf(format!(
228            "font cell {cell_w}x{cell_h} exceeds the 255px bitmap font limit"
229        )));
230    }
231
232    let first = *selected.first().unwrap();
233    let last = *selected.last().unwrap();
234    let bytes_per_row = (cell_w as u32).div_ceil(8) as usize;
235    let glyph_len = bytes_per_row * cell_h as usize;
236    let mut out = vec![0u8; glyph_len * (last - first + 1) as usize];
237    let mut advance = 0i32;
238
239    for code in selected {
240        let glyph = &glyphs[&code];
241        advance = advance.max(if glyph.dwidth > 0 {
242            glyph.dwidth
243        } else {
244            glyph.bbx.0
245        });
246        let (gw, gh, gx, gy) = glyph.bbx;
247        // BDF y offsets are measured up from the baseline; our rows run down
248        // from the cell top. Keeping every glyph relative to the shared cell
249        // preserves how they line up on the baseline.
250        let dx = gx - left;
251        let dy = top - (gy + gh);
252        let base = glyph_len * (code - first) as usize;
253        for (row_idx, row) in glyph.rows.iter().enumerate() {
254            let ty = dy + row_idx as i32;
255            if ty < 0 || ty >= cell_h {
256                continue;
257            }
258            for col in 0..gw {
259                let byte = (col / 8) as usize;
260                let bit = 7 - (col % 8);
261                let set = row.get(byte).is_some_and(|b| b & (1 << bit) != 0);
262                if !set {
263                    continue;
264                }
265                let tx = dx + col;
266                if tx < 0 || tx >= cell_w {
267                    continue;
268                }
269                let idx = base + ty as usize * bytes_per_row + (tx / 8) as usize;
270                out[idx] |= 0x80 >> (tx % 8);
271            }
272        }
273    }
274
275    Ok(BitmapFontData {
276        width: cell_w as u8,
277        height: cell_h as u8,
278        advance: advance.clamp(1, 255) as u8,
279        line_height: cell_h as u8,
280        first_char: first as u8,
281        bytes_per_row: bytes_per_row as u8,
282        glyphs: out,
283    })
284}

286fn hex_row(line: &str) -> Option<Vec<u8>> {
287    if !line.len().is_multiple_of(2) {
288        return None;
289    }
290    let mut bytes = Vec::with_capacity(line.len() / 2);
291    for pair in line.as_bytes().chunks(2) {
292        let s = std::str::from_utf8(pair).ok()?;
293        bytes.push(u8::from_str_radix(s, 16).ok()?);
294    }
295    Some(bytes)
296}

298fn read_ints<'a>(parts: impl Iterator<Item = &'a str>, count: usize) -> Option<Vec<i32>> {
299    let values: Vec<i32> = parts.take(count).filter_map(|v| v.parse().ok()).collect();
300    (values.len() == count).then_some(values)
301}

303/// Triangle mesh in the layout `embedded_3dgfx::Geometry` expects.
304#[derive(Clone, Debug, Default, PartialEq)]
305pub struct MeshData {
306    pub vertices: Vec<[f32; 3]>,
307    pub faces: Vec<[usize; 3]>,
308    /// One normal per face, for flat-shaded lighting.
309    pub normals: Vec<[f32; 3]>,
310}
311
312/// Parses the subset of Wavefront OBJ that matters for decorative geometry:
313/// `v` positions and `f` faces, with polygons fan-triangulated.
314pub fn parse_obj(source: &str) -> Result<MeshData, AssetError> {
315    let mut vertices: Vec<[f32; 3]> = Vec::new();
316    let mut faces: Vec<[usize; 3]> = Vec::new();
317
318    for line in source.lines() {
319        let line = line.trim();
320        let mut parts = line.split_whitespace();
321        match parts.next() {
322            Some("v") => {
323                let coords: Vec<f32> = parts.take(3).filter_map(|v| v.parse().ok()).collect();
324                if coords.len() != 3 {
325                    return Err(AssetError::Obj(format!("malformed vertex '{line}'")));
326                }
327                vertices.push([coords[0], coords[1], coords[2]]);
328            }
329            Some("f") => {
330                let indices: Vec<usize> = parts
331                    .filter_map(|token| {
332                        let first = token.split('/').next()?;
333                        let idx: i64 = first.parse().ok()?;
334                        // OBJ indices are 1-based; negatives count back from the end.
335                        let resolved = if idx < 0 {
336                            vertices.len() as i64 + idx
337                        } else {
338                            idx - 1
339                        };
340                        (resolved >= 0 && (resolved as usize) < vertices.len())
341                            .then_some(resolved as usize)
342                    })
343                    .collect();
344                for i in 1..indices.len().saturating_sub(1) {
345                    faces.push([indices[0], indices[i], indices[i + 1]]);
346                }
347            }
348            _ => {}
349        }
350    }
351
352    if vertices.is_empty() || faces.is_empty() {
353        return Err(AssetError::Empty("OBJ mesh"));
354    }
355
356    let mut mesh = MeshData {
357        vertices,
358        faces,
359        normals: Vec::new(),
360    };
361    mesh.recompute_normals();
362    Ok(mesh)
363}

365/// Parses a mesh from file bytes, choosing the reader by extension.
366///
367/// CAD tools export STL and modelling tools export OBJ, so a project can point
368/// a `mesh` node at whichever file the artist already has.
369pub fn parse_mesh(source_name: &str, bytes: &[u8]) -> Result<MeshData, AssetError> {
370    let extension = source_name
371        .rsplit('.')
372        .next()
373        .unwrap_or_default()
374        .to_ascii_lowercase();
375    match extension.as_str() {
376        "obj" => {
377            let text = std::str::from_utf8(bytes)
378                .map_err(|err| AssetError::Obj(format!("not UTF-8 text: {err}")))?;
379            parse_obj(text)
380        }
381        "stl" => parse_stl(bytes),
382        _ => Err(AssetError::UnsupportedMesh(source_name.to_string())),
383    }
384}

386/// Parses binary or ASCII STL.
387///
388/// STL repeats full coordinates per triangle corner, so identical positions are
389/// welded back into shared vertices; the 3D rasterizer transforms each vertex
390/// once, and a logo exported from CAD typically sheds two thirds of its list.
391pub fn parse_stl(bytes: &[u8]) -> Result<MeshData, AssetError> {
392    let triangles = if is_ascii_stl(bytes) {
393        let text = std::str::from_utf8(bytes)
394            .map_err(|err| AssetError::Stl(format!("not UTF-8 text: {err}")))?;
395        parse_ascii_stl(text)?
396    } else {
397        parse_binary_stl(bytes)?
398    };
399
400    let mut vertices: Vec<[f32; 3]> = Vec::new();
401    let mut faces: Vec<[usize; 3]> = Vec::new();
402    let mut lookup: BTreeMap<[u32; 3], usize> = BTreeMap::new();
403
404    for corners in triangles {
405        let mut face = [0usize; 3];
406        for (slot, corner) in corners.iter().enumerate() {
407            let key = [
408                corner[0].to_bits(),
409                corner[1].to_bits(),
410                corner[2].to_bits(),
411            ];
412            face[slot] = *lookup.entry(key).or_insert_with(|| {
413                vertices.push(*corner);
414                vertices.len() - 1
415            });
416        }
417        // Drop degenerate triangles: they carry no surface and would only
418        // produce a zero-length normal.
419        if face[0] != face[1] && face[1] != face[2] && face[0] != face[2] {
420            faces.push(face);
421        }
422    }
423
424    if vertices.is_empty() || faces.is_empty() {
425        return Err(AssetError::Empty("STL mesh"));
426    }
427
428    let mut mesh = MeshData {
429        vertices,
430        faces,
431        normals: Vec::new(),
432    };
433    mesh.recompute_normals();
434    Ok(mesh)
435}

437fn is_ascii_stl(bytes: &[u8]) -> bool {
438    // The 80-byte binary header may itself start with "solid", so trust the
439    // triangle count instead: a binary file is exactly 84 + 50 * count bytes.
440    if bytes.len() >= 84 {
441        let count = u32::from_le_bytes([bytes[80], bytes[81], bytes[82], bytes[83]]) as usize;
442        if bytes.len() == 84 + count * 50 {
443            return false;
444        }
445    }
446    bytes.starts_with(b"solid")
447}
448
449fn parse_binary_stl(bytes: &[u8]) -> Result<Vec<[[f32; 3]; 3]>, AssetError> {
450    if bytes.len() < 84 {
451        return Err(AssetError::Stl("file is shorter than the header".into()));
452    }
453    let count = u32::from_le_bytes([bytes[80], bytes[81], bytes[82], bytes[83]]) as usize;
454    let expected = 84 + count * 50;
455    if bytes.len() < expected {
456        return Err(AssetError::Stl(format!(
457            "header claims {count} triangles but the file holds {} bytes",
458            bytes.len()
459        )));
460    }
461
462    let mut triangles = Vec::with_capacity(count);
463    for index in 0..count {
464        // Each record is a face normal we recompute anyway, then 3 corners.
465        let base = 84 + index * 50 + 12;
466        let mut corners = [[0.0f32; 3]; 3];
467        for (corner_idx, corner) in corners.iter_mut().enumerate() {
468            for (axis, value) in corner.iter_mut().enumerate() {
469                let at = base + corner_idx * 12 + axis * 4;
470                *value =
471                    f32::from_le_bytes([bytes[at], bytes[at + 1], bytes[at + 2], bytes[at + 3]]);
472            }
473        }
474        triangles.push(corners);
475    }
476    Ok(triangles)
477}

479fn parse_ascii_stl(source: &str) -> Result<Vec<[[f32; 3]; 3]>, AssetError> {
480    let mut triangles = Vec::new();
481    let mut corners: Vec<[f32; 3]> = Vec::new();
482
483    for line in source.lines() {
484        let mut parts = line.split_whitespace();
485        match parts.next() {
486            Some("vertex") => {
487                let coords: Vec<f32> = parts.take(3).filter_map(|v| v.parse().ok()).collect();
488                if coords.len() != 3 {
489                    return Err(AssetError::Stl(format!(
490                        "malformed vertex '{}'",
491                        line.trim()
492                    )));
493                }
494                corners.push([coords[0], coords[1], coords[2]]);
495            }
496            Some("endloop") => {
497                if corners.len() == 3 {
498                    triangles.push([corners[0], corners[1], corners[2]]);
499                }
500                corners.clear();
501            }
502            _ => {}
503        }
504    }
505    Ok(triangles)
506}

508impl MeshData {
509    pub fn recompute_normals(&mut self) {
510        self.normals.clear();
511        self.normals.reserve(self.faces.len());
512        for [a, b, c] in self.faces.iter().copied() {
513            let (va, vb, vc) = (self.vertices[a], self.vertices[b], self.vertices[c]);
514            let u = [vb[0] - va[0], vb[1] - va[1], vb[2] - va[2]];
515            let v = [vc[0] - va[0], vc[1] - va[1], vc[2] - va[2]];
516            let n = [
517                u[1] * v[2] - u[2] * v[1],
518                u[2] * v[0] - u[0] * v[2],
519                u[0] * v[1] - u[1] * v[0],
520            ];
521            let len = (n[0] * n[0] + n[1] * n[1] + n[2] * n[2]).sqrt();
522            self.normals.push(if len > f32::EPSILON {
523                [n[0] / len, n[1] / len, n[2] / len]
524            } else {
525                [0.0, 0.0, 1.0]
526            });
527        }
528    }
529
530    /// Centers the mesh on the origin and scales it to unit radius, so the
531    /// `scale` authored in KDL means the same thing for any source model.
532    pub fn normalize(&mut self) {
533        if self.vertices.is_empty() {
534            return;
535        }
536        let mut min = [f32::MAX; 3];
537        let mut max = [f32::MIN; 3];
538        for v in &self.vertices {
539            for axis in 0..3 {
540                min[axis] = min[axis].min(v[axis]);
541                max[axis] = max[axis].max(v[axis]);
542            }
543        }
544        let center = [
545            (min[0] + max[0]) / 2.0,
546            (min[1] + max[1]) / 2.0,
547            (min[2] + max[2]) / 2.0,
548        ];
549        let mut radius = 0.0f32;
550        for v in &self.vertices {
551            let d = [v[0] - center[0], v[1] - center[1], v[2] - center[2]];
552            radius = radius.max((d[0] * d[0] + d[1] * d[1] + d[2] * d[2]).sqrt());
553        }
554        if radius <= f32::EPSILON {
555            return;
556        }
557        for v in &mut self.vertices {
558            for axis in 0..3 {
559                v[axis] = (v[axis] - center[axis]) / radius;
560            }
561        }
562    }
563}

659        }

690    #[test]
691    fn triangulates_obj_quads_and_normalizes() {
692        let obj = "\
693v -1 -1 0
694v 1 -1 0
695v 1 1 0
696v -1 1 0
697f 1 2 3 4
698";
699        let mut mesh = parse_obj(obj).unwrap();
700        assert_eq!(mesh.faces, vec![[0, 1, 2], [0, 2, 3]]);
701        assert_eq!(mesh.normals.len(), 2);
702        assert!((mesh.normals[0][2] - 1.0).abs() < 1e-5);
703        mesh.normalize();
704        let radius = mesh
705            .vertices
706            .iter()
707            .map(|v| (v[0] * v[0] + v[1] * v[1] + v[2] * v[2]).sqrt())
708            .fold(0.0f32, f32::max);
709        assert!((radius - 1.0).abs() < 1e-5);
710    }
